    def create_progress_bars() -> tuple[Progress, Progress]:
    """
    Create progress bars for the layout.

    Returns:
        Tuple of (overall_progress, chunk_progress) Progress instances
    """
    overall_progress = Progress(
        "[bold blue]{task.description}",
        BarColumn(),
        TaskProgressColumn(),
        TimeElapsedColumn(),
        TextColumn("[bold green]{task.completed}/{task.total}"),
    )

    chunk_progress = Progress(
        "[bold green]Processing Chunk:",
        BarColumn(),
        TaskProgressColumn(),
        TimeElapsedColumn(),
        TextColumn("Records Processed: [bold green]{task.completed}/{task.total}"),
    )

    return overall_progress, chunk_progress

    def generate_chunk(chunk_size: int, progress: Progress, task_id: int) -> pd.DataFrame:
    """
    Generate a chunk of data with progress tracking.

    Args:
        chunk_size: Number of records to generate
        progress: Rich Progress instance for tracking
        task_id: ID of the task in the progress bar

    Returns:
        DataFrame containing the generated chunk
    """
    progress.start_task(task_id)

    # Determine optimal subchunk size based on available memory
    mem = psutil.virtual_memory()
    available_mem_gb = mem.available / (1024 ** 3)
    records_per_gb = 100000  # Approximate number of records that fit in 1GB

    subchunk_size = min(chunk_size, int(available_mem_gb * records_per_gb / 2))
    subchunk_size = max(1000, subchunk_size)  # Ensure minimum size

    num_subchunks = (chunk_size + subchunk_size - 1) // subchunk_size
    results = []

    for i in range(num_subchunks):
        actual_size = min(subchunk_size, chunk_size - i * subchunk_size)
        sub_chunk = generate_subchunk(actual_size)
        results.append(sub_chunk)
        progress.update(task_id, advance=actual_size)

    return pd.concat([pd.DataFrame(sub_chunk) for sub_chunk in results], ignore_index=True)

    def generate_data_parallel_with_panels(
    num_records: int, 
    chunk_size: int, 
    max_workers: Optional[int] = None
) -> pd.DataFrame:
    """
    Generate data in parallel with Rich layout, panels, and progress tracking.

    Args:
        num_records: Total number of records to generate
        chunk_size: Number of records per chunk
        max_workers: Maximum number of worker threads (defaults to 5x physical cores)

    Returns:
        DataFrame containing all generated records
    """
    chunks = num_records // chunk_size
    remainder = num_records % chunk_size
    total_tasks = chunks + (1 if remainder else 0)
    chunk_status = {"waiting": chunks, "running": 0, "completed": 0}

    # Layout configuration
    layout = Layout()
    layout.split_row(
        Layout(name="left", ratio=1),
        Layout(name="middle", ratio=1),
        Layout(name="right", ratio=1),
    )

    overall_progress, chunk_progress = create_progress_bars()
    overall_task = overall_progress.add_task("[cyan]Overall Progress", total=total_tasks)

    # Set up the layout panels
    layout["left"].update(Panel(overall_progress, title="Overall Progress"))
    layout["middle"].update(Panel(chunk_progress, title="Chunk Progress"))

    if max_workers is None:
        num_physical_cores = psutil.cpu_count(logical=False)
        max_workers = num_physical_cores * 5

    with ThreadPoolExecutor(max_workers=max_workers) as executor, Live(layout, refresh_per_second=10):
        futures = []
        futures_to_tasks = {}
        task_queue = queue.Queue()
        # Create the stop event
        stop_event = threading.Event()

        system_vitals_thread = threading.Thread(
            target=update_system_vitals, args=(layout, stop_event), daemon=True
        )
        system_vitals_thread.start()

        for i in range(chunks):
            chunk_status["waiting"] -= 1
            chunk_task = chunk_progress.add_task(f"Chunk {i + 1}", total=chunk_size, start=False)
            future = executor.submit(generate_chunk, chunk_size, chunk_progress, chunk_task)
            futures.append(future)
            futures_to_tasks[future] = chunk_task

        if remainder:
            chunk_status["waiting"] -= 1
            chunk_task = chunk_progress.add_task(f"Chunk {chunks + 1}", total=remainder, start=False)
            future = executor.submit(generate_chunk, remainder, chunk_progress, chunk_task)
            futures.append(future)
            futures_to_tasks[future] = chunk_task

        # Process results as they complete
        dataframes = []
        for future in as_completed(futures):
            try:
                df = future.result()
                dataframes.append(df)
                overall_progress.update(overall_task, advance=1)
            except Exception as e:
                print(f"Error in chunk: {str(e)}")
                traceback.print_exc()

        # Signal the system vitals thread to stop
        stop_event.set()
        system_vitals_thread.join(timeout=1.0)

    # Combine all dataframes
    return pd.concat(dataframes, ignore_index=True)

    def update_system_vitals(layout: Layout, stop_event: threading.Event, interval: float = 1):
    """
    Update system vitals in the right panel of the layout.

    Args:
        layout: Rich Layout instance
        stop_event: Threading event to signal when to stop
        interval: Update interval in seconds
    """
    while not stop_event.is_set():
        # Create a table to display system vitals
        vitals_table = Table(title="System Vitals", show_header=True)
        vitals_table.add_column("Metric", justify="left")
        vitals_table.add_column("Value", justify="right")

        # Collect CPU, memory, and thread utilization data
        cpu_percent = psutil.cpu_percent(interval=0)
        memory_info = psutil.virtual_memory()
        disk_usage = psutil.disk_usage('/')
        net_io = psutil.net_io_counters()
        thread_count = threading.active_count()

        # Calculate network throughput (bytes sent/received since last check)
        net_io_sent = net_io.bytes_sent / (1024 ** 2)  # Convert to MB
        net_io_recv = net_io.bytes_recv / (1024 ** 2)  # Convert to MB

        # Add data to the table
        vitals_table.add_row("CPU Usage", f"{cpu_percent}%")
        vitals_table.add_row("Memory Usage", f"{memory_info.percent}%")
        vitals_table.add_row("Threads", str(thread_count))
        vitals_table.add_row("Disk Usage", f"{disk_usage.percent}%")
        vitals_table.add_row("Network Sent", f"{net_io_sent:.2f} MB")
        vitals_table.add_row("Network Received", f"{net_io_recv:.2f} MB")

        # Display the table in the right panel of the layout
        layout["right"].update(Panel(vitals_table, title="System Health"))

        # Refresh interval
        time.sleep(interval)

    def write_output_file(df: pd.DataFrame, output_file: str, separator: Optional[str] = None):
    """
    Write DataFrame to output file in fixed-length or separator-delimited format.

    Args:
        df: DataFrame to write
        output_file: Path to output file
        separator: If provided, use this separator for delimited format; otherwise use fixed-length
    """
    if separator:
        # Write a separator-delimited file (e.g., CSV-like)
        df.to_csv(output_file, sep=separator, index=False, encoding='utf-8')
    else:
        overall_progress = Progress(
            "[bold blue]{task.description}",
            BarColumn(),
            TaskProgressColumn(),
            TimeElapsedColumn(),
            TextColumn("[bold green]{task.completed}/{task.total}"),
        )
        overall_task = overall_progress.add_task("[cyan]Writing fixed-length file", total=df.shape[0])

        # Write fixed-length format
        with open(output_file, 'w', encoding='utf-8') as f:
            with Live(overall_progress, refresh_per_second=10):
                for row in df.itertuples(index=False):
                    fixed_length_record = (
                        f"{str(row.ICA).zfill(11)}"
                        f"{row.PAN: >19}"
                        f"{row.Current_PAN_Status: <1}"
                        f"{row.Linkage_Number: <19}"
                        f"{row.Group_ID.upper(): <25}"
                        f"{row.PAN_Total_Spend: <14}"
                        f"{row.Accumulative_Spend_from_Linked_PANs: <14}"
                        # Add remaining fields...
                    )
                    f.write(fixed_length_record + '\n')
                    overall_progress.update(overall_task, advance=1)
